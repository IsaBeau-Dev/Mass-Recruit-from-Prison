on_game_start = {
    on_actions = {
        on_mrfp_start
    }
}
on_mrfp_start = {
    effect = {
        set_global_variable = {
            name = mrfp_is_loaded
            value = yes
        }
    }
}


on_game_start_after_lobby = {
	on_actions = { 
        mrfp_start_story
		mrfp_welcome_event
	}
}

mrfp_start_story = {
    effect = {
        every_player = {
            create_story = mrfp_recruitment_cycle
        }
    }
}

mrfp_welcome_event = {
	effect = {
        if = {
            limit = { has_game_rule = on_mrfp_open_settings_on_new_game }
            every_player = {
                save_scope_as = actor
                scope:actor.var:mrfp_story = {
                    save_scope_as = the_story
                }
                if = {
                    limit = { has_game_rule = mrfp_new_and_old_settings }
                    trigger_event = mrfp_set_threshold.0001
                }
                else = {
                    #open settings via workaround with character flag
                    add_character_flag = mrfp_settings_window_show

                }
                # trigger_event = mrfp_welcome_events.0001
            }
        }
	}
}


on_release_from_prison = {
    on_actions = {
        mrfp_on_release_from_prison
    }
}

mrfp_on_release_from_prison = {
    effect = {
        if = {
            limit = { has_character_flag = mrfp_blocked }
            remove_character_flag = mrfp_blocked
        }
        if = {
            limit = { has_character_flag = mrfp_blocked_gui }
            remove_character_flag = mrfp_blocked_gui
        }
        if = {
            limit = { has_character_flag = mrfp_blocked_can_ransom }
            remove_character_flag = mrfp_blocked_can_ransom
        }
    }
}

on_death = {
    on_actions = {
        mrfp_on_death
        mrfp_on_death_ransom
        #mrfp_on_death_vars
    }
}

mrfp_on_death = {
    trigger = {
        has_character_flag = enable_auto_mrfp
    }
    effect = {
        primary_heir = {
            add_character_flag = enable_auto_mrfp
        }
    }
}
mrfp_on_death_ransom = {
    trigger = {
        has_character_flag = enable_auto_mrfp_ransom
    }
    effect = {
        primary_heir = {
            add_character_flag = enable_auto_mrfp_ransom
        }
    }
}


### Called when player has confirmed their selection of a destiny character
# Scopes:
# root: destiny character (is already current player)
# scope:destiny_type_flag - flag of selected destiny type
# scope:previous_player_character - previous player character
# scope:previous_player_heir - previous player heir

on_player_select_destiny_confirmed = {
    on_actions = {
        mrfp_recruit_destiny_confirmed
        mrfp_ransom_destiny_confirmed
        mrfp_story_transfer
    }
}

mrfp_recruit_destiny_confirmed = {
    trigger = {
        scope:previous_player_heir = {
            has_character_flag = enable_auto_mrfp
        }
    }
    effect = {
        scope:previous_player_heir = {
            remove_character_flag = enable_auto_mrfp
        }
        add_character_flag = enable_auto_mrfp
    }
}

mrfp_ransom_destiny_confirmed = {
    trigger = {
        scope:previous_player_heir = {
            has_character_flag = enable_auto_mrfp_ransom
        }
    }
    effect = {
        scope:previous_player_heir = {
            remove_character_flag = enable_auto_mrfp_ransom
        }
        add_character_flag = enable_auto_mrfp_ransom
    }
}

mrfp_story_transfer = {
    trigger = {
        scope:previous_player_heir = {
            has_variable = mrfp_story
        }
    }
    effect = {
        #transfer story ownership
        scope:previous_player_heir.var:mrfp_story = {
            make_story_owner = root
        }

        #transfer story variable to new owner
        set_variable = {
            name = mrfp_story
            value = scope:previous_player_heir.var:mrfp_story
        }
        
        #remove variable from old owner
        scope:previous_player_heir = {
            remove_variable = mrfp_story
        }
    }
}

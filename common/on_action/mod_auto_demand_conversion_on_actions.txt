quarterly_playable_pulse = {
    on_actions = {
        mod_auto_demand_conversion_on_action
        delay = { months = 1 }
        mod_auto_demand_conversion_on_action
        delay = { months = 2 }
        mod_auto_demand_conversion_on_action
    }
}

mod_auto_demand_conversion_on_action = {
    trigger = {
        has_character_flag = enable_auto_demand_conversion
    }
    effect = {
        house = {
            every_house_member = {
                save_scope_as = target
                mrfp_run_interaction_if_possible = { INTERACTION_KEY = demand_conversion_interaction }
                clear_saved_scope = target
            }
        }

        every_vassal_or_below = {
            save_scope_as = target
            mrfp_run_interaction_if_possible = { INTERACTION_KEY = demand_conversion_vassal_ruler_interaction }
            clear_saved_scope = target
        }

        every_courtier = {
            save_scope_as = target
            mrfp_run_interaction_if_possible = { INTERACTION_KEY = ask_for_conversion_courtier_interaction }
            clear_saved_scope = target
        }
    }
}

on_death = {
    on_actions = {
        mod_auto_demand_conversion_on_death
    }
}
mod_auto_demand_conversion_on_death = {
    trigger = {
        has_character_flag = enable_auto_demand_conversion
    }
    effect = {
        primary_heir = {
            add_character_flag = enable_auto_demand_conversion
        }
        
    }
}

quarterly_playable_pulse = {
    on_actions = {
        mod_auto_demand_conversion_on_action
        delay = { months = 1 }
        mod_auto_demand_conversion_on_action
        delay = { months = 2 }
        mod_auto_demand_conversion_on_action
    }
}

mod_auto_demand_conversion_on_action = {
    trigger = {
        has_character_flag = enable_auto_demand_conversion
    }
    effect = {
        house = {
            every_house_member = {
                if = {
                    limit = {
                        is_character_interaction_potentially_accepted = {
                            interaction = demand_conversion_interaction
                            recipient = this
                            required_response = yes
                        }
                    }
                }
                run_interaction = {
                    interaction = demand_conversion_interaction
                    actor = root
                    recipient = this
                    send_threshold = maybe
                }
            }
        }

        every_vassal_or_below = {
            if = {
                limit = {
                    is_character_interaction_potentially_accepted = {
                        interaction = demand_conversion_vassal_ruler_interaction
                        recipient = this
                        required_response = yes
                    }
                }
            }
            run_interaction = {
                interaction = demand_conversion_vassal_ruler_interaction
                actor = root
                recipient = this
                send_threshold = maybe
            }
        }

        every_courtier = {
            if = {
                limit = {
                    is_character_interaction_potentially_accepted = {
                        interaction = ask_for_conversion_courtier_interaction
                        recipient = this
                        required_response = yes
                    }
                }
            }
            run_interaction = {
                interaction = ask_for_conversion_courtier_interaction
                actor = root
                recipient = this
                send_threshold = maybe
            }
        }
    }
}

on_death = {
    on_actions = {
        mod_auto_demand_conversion_on_death
    }
}
mod_auto_demand_conversion_on_death = {
    trigger = {
        has_character_flag = enable_auto_demand_conversion
    }
    effect = {
        primary_heir = {
            add_character_flag = enable_auto_demand_conversion
        }
        
    }
}

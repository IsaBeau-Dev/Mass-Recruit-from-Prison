mrfp_secondary_recipient_gold = {
    scope:target = {
        save_temporary_scope_as = secondary_recipient

        if = {
            limit = {
                is_ruler = no
                exists = liege
            }
            liege = {
                save_temporary_scope_as = secondary_recipient
            }
        }
    }
    value = scope:secondary_recipient.gold
}

mrfp_secondary_recipient_herd = {
    scope:target = {
        save_temporary_scope_as = secondary_recipient

        if = {
            limit = {
                is_ruler = no
                exists = liege
            }
            liege = {
                save_temporary_scope_as = secondary_recipient
            }
        }
    }
    value = scope:secondary_recipient.domicile.herd
    round = yes
}

mrfp_target_ransom_gold_cost = {
    value = scope:target.ransom_cost_value
    if = {
        limit = {
                exists = dynasty
                dynasty = {
                    has_dynasty_perk = fp1_pillage_legacy_3
                }
        }
        value = scope:target.increased_ransom_cost_value
    }
}


#prowess limit
mrfp_settings_prowess_limit_toggle = {
  scope = character
  effect = {
    mrfp_start_story_if_not_exist = yes
    root.var:mrfp_story = {
      save_scope_as = the_story
    }
    mrfp_toggle_variable = { VARIABLE_NAME = mrfp_threshold_enabled }
  }
}

mrfp_settings_prowess_limit_add_1 = {
  scope = character
  effect = {
    mrfp_start_story_if_not_exist = yes
    save_scope_value_as = { name = mrfp_prowess_threshold_add_or_subtract value = 1 }
    root.var:mrfp_story = {
      set_variable = {
        name = mrfp_prowess_threshold
        value = mrfp_prowess_threshold_script_value
      }
    }
  }
}

### Monthly actions
mrfp_settings_toggle_monthly_recruit = {
  scope = character
  effect = {
    mrfp_start_story_if_not_exist = yes
    if = {
      limit = { has_character_flag = enable_auto_mrfp }
      remove_character_flag = enable_auto_mrfp
    }
    else = {
      add_character_flag = enable_auto_mrfp
    }
  }
}
mrfp_settings_toggle_monthly_ransom = {
  scope = character
  effect = {
    mrfp_start_story_if_not_exist = yes
    if = {
      limit = { has_character_flag = enable_auto_mrfp_ransom }
      remove_character_flag = enable_auto_mrfp_ransom
    }
    else = {
      add_character_flag = enable_auto_mrfp_ransom
    }
  }
}
